cmake_minimum_required(VERSION 3.22)
project(ego_gazebo)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(dependencies rclcpp px4_msgs quadrotor_msgs geometry_msgs nav_msgs sensor_msgs std_msgs std_srvs tf2_ros)
foreach(dependency IN LISTS dependencies)
  find_package(${dependency} REQUIRED)
endforeach()
foreach(node sim_adapter offboard verify_sim)
  add_executable(${node} src/${node}.cpp)
  target_include_directories(${node} PRIVATE include)
  target_link_libraries(${node} Eigen3::Eigen)
  target_compile_options(${node} PRIVATE -Wall -Wextra -Wpedantic)
  ament_target_dependencies(${node} ${dependencies})
  install(TARGETS ${node} DESTINATION lib/${PROJECT_NAME})
endforeach()
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(DIRECTORY ../../assets/ DESTINATION share/${PROJECT_NAME}/assets)
install(PROGRAMS ../../scripts/run_px4.sh DESTINATION lib/${PROJECT_NAME})
ament_package()
